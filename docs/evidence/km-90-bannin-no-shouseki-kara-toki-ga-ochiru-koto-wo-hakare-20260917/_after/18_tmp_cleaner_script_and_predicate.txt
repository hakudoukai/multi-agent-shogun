     1	#!/bin/sh
     2	#
     3	# Perform temporary directory cleaning so that long-lived systems
     4	# don't end up with excessively old files there.
     5	#
     6
     7	# Configurations
     8	daily_clean_tmps_dirs="/tmp"				# Delete under here
     9	daily_clean_tmps_days="3"				# If not accessed for
    10	daily_clean_tmps_ignore=".X*-lock .X11-unix .ICE-unix .font-unix .XIM-unix"
    11	daily_clean_tmps_ignore="$daily_clean_tmps_ignore quota.user quota.group"
    12								# Don't delete these
    13
    14	echo "Removing old temporary files:"
    15
    16	set -f noglob
    17	args="-atime +$daily_clean_tmps_days -mtime +$daily_clean_tmps_days"
    18	args="${args} -ctime +$daily_clean_tmps_days"
    19	dargs="-empty -mtime +$daily_clean_tmps_days"
    20	dargs="${dargs} ! -name .vfs_rsrc_streams_*"
    21	args="$args "`echo " ${daily_clean_tmps_ignore% }" |
    22	    sed 's/[ 	][ 	]*/ ! -name /g'`
    23	dargs="$dargs "`echo " ${daily_clean_tmps_ignore% }" |
    24	    sed 's/[ 	][ 	]*/ ! -name /g'`
    25
    26	for dir in $daily_clean_tmps_dirs
    27	do
    28	    [ ."${dir#/}" != ."$dir" -a -d $dir ] && cd $dir && {
    29	    find -dx . -fstype local -type f $args -delete -print
    30	    find -dx . -fstype local ! -name . -type d $dargs -delete -print
    31	    } | sed "s,^\\.,  $dir,"
    32	done
    33	set -f glob
    34
    35	exit 0
