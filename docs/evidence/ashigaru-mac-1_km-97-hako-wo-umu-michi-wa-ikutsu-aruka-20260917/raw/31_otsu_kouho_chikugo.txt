class InboxLock:
    def __init__(self, path, timeout=5.0):
        self.d = path + '.lock.d'; self.timeout = timeout; self.held = False
    def __enter__(self):
        t0 = time.time()
        while True:
            try:
                os.mkdir(self.d); self.held = True; return self

                if time.time() - t0 > self.timeout:

                time.sleep(0.05)
    def __exit__(self, *a):
        if self.held:
            try: os.rmdir(self.d)
            except OSError: pass
        return False
