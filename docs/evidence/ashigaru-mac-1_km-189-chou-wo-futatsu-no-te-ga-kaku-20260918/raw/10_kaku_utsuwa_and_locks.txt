# 本機は flock 缺(實測) ∴ inbox_write.sh の實經路は mkdir("<inbox>.lock.d")。
# ★共有器は不觸★。同じ約束を★此方が守る★だけである。取れねば★印を打たぬ(fail-closed)★。
import os, time
class InboxLock:
    def __init__(self, path, timeout=5.0):
        self.d = path + '.lock.d'; self.timeout = timeout; self.held = False
    def __enter__(self):
        t0 = time.time()
        while True:
            try:
                os.mkdir(self.d); self.held = True; return self
            except FileExistsError:
                if time.time() - t0 > self.timeout:
                    raise TimeoutError('lock 取得できず: ' + self.d)
                time.sleep(0.05)
    def __exit__(self, *a):
        if self.held:
            try: os.rmdir(self.d)
            except OSError: pass
        return False
